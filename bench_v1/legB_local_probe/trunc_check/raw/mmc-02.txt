import json
import sys
import random
import math
from datetime import datetime

def main():
    import argparse
    parser = argparse.ArgumentParser()
    parser.add_argument('--seeds', type=int, default=10)
    parser.add_argument('--horizon', type=float, default=100)
    parser.add_argument('--seed-start', type=int, default=0)
    parser.add_argument('--pert', type=str, default='none')
    args = parser.parse_args()

    time_unit = 'minute'
    runtime_units = 'minute'
    win = args.horizon
    seeds_used = args.seeds
    seed_list = [args.seed_start + i for i in range(seeds_used)]
    per_seed_L = []
    per_seed_W = []
    per_seed_busy_frac = []

    for seed in seed_list:
        random.seed(seed)
        if args.pert == 'lam_up':
            lam = 8 * 1.1
        elif args.pert == 'mu_up':
            mu = 10 * 1.1
        elif args.pert == 'c_up':
            c = 2 * 1.1
        else:
            lam = 8
            mu = 10
            c = 2

        events = []
        current_time = 0
        queue = []
        busy = 0
        busy_time = 0
        total_time = 0
        total_L = 0
        total_W = 0
        total_arr = 0
        total_blk = 0
        total_dep = 0
        total_dep_rate = 0
        total_busy_frac = 0
        total_p_loss = 0
        total_p_wait = 0
        start_time = None
        end_time = None

        while current_time < win:
            if not events:
                next_arrival = random.expovariate(lam)
                events.append((current_time + next_arrival, 'arrival'))
                current_time = current_time + next_arrival
            else:
                next_event = min(events)
                event_time, event_type = next_event
                if event_time > win:
                    break
                if event_type == 'arrival':
                    total_arr += 1
                    service_time = random.expovariate(mu)
                    if busy < c:
                        busy += 1
                        events.append((event_time + service_time, 'departure'))
                        total_busy_frac += 1
                    else:
                        queue.append(service_time)
                        total_p_wait += 1
                elif event_type == 'departure':
                    total_blk += 1
                    total_dep += 1
                    if busy > 0:
                        busy -= 1
                        if queue:
                            service_time = queue.pop(0)
                            events.append((event_time + service_time, 'departure'))
                            total_busy_frac += 1
                    else:
                        total_p_loss += 1
                current_time = event_time

        if start_time is None:
            start_time = current_time
        end_time = current_time
        total_time = end_time - start_time
        total_L = total_busy_frac / total_time
        total_W = total_W / total_time
        total_busy_frac = total_busy_frac / total_time
        total_p_wait = total_p_wait / total_time
        total_p_loss = total_p_loss / total_time
        total_dep_rate = total_dep / total_time
        per_seed_L.append(total_L)
        per_seed_W.append(total_W)
        per_seed_busy_frac.append(total_busy_frac)

    L = sum(per_seed_L) / len(per_seed_L)
    W = sum(per_seed_W) / len(per_seed_W)
    busy_frac = sum(per_seed_busy_frac) / len(per_seed_busy_frac)
    arr = total_arr / total_time
    blk = total_blk / total_time
    dep = total_dep / total_time
    dep_rate = total_dep_rate
    p_loss = total_p_loss
    p_wait = total_p_wait
    result = {
        "L": L,
        "W": W,
        "arr": arr,
        "blk": blk,
        "busy_frac": busy_frac,
        "dep": dep,
        "dep_rate": dep_rate,
        "p_loss": p_loss,
        "p_wait": p_wait,
        "per_seed_L": per_seed_L,
        "per_seed_W": per_seed_W,
        "per_seed_busy_frac": per_seed_busy_frac,
        "runtime_units": runtime_units,
        "seed_list": seed_list,
        "seeds_used": seeds_used,
        "time_unit": time_unit,
        "win": win
    }
    print(json.dumps(result))

if __name__ == "__main__":
    main()
